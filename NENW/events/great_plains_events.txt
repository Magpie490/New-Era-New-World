# Invasion setup
character_event = {
    id = gplains.1

    hide_window = yes
    is_triggered_only = yes

	only_playable = yes
	religion = catholic
	min_age = 16
    
	trigger = {
		OR = {
			year = 2500
			AND = {
				has_game_rule = {
					name = mongol_invasion
					value = random
				}
				total_years_played = 50
			}
			AND = {
				has_game_rule = {
					name = mongol_invasion
					value = delayed_random
				}
				year = 2400
				total_years_played = 50
			}
		}
		NOR = { 
			year = 2550
			has_game_rule = {
				name = mongol_invasion
				value = off
			}
		}
		controls_religion = yes # Let it happen for the Pope. He's always around.
		NOT = {
			has_global_flag = joshua_arrival
		}
	}
	
	weight_multiplier = {
		days = 1
		modifier = {
			factor = 1.5
			year = 2505
		}
		modifier = {
			factor = 5
			year = 2510
		}
		modifier = {
			factor = 100 #If game has historical rule, ensure that the Mongol Invasion WILL happen when the time comes.
			has_game_rule = {
				name = mongol_invasion
				value = historical
			}
		}
		modifier = {
			factor = 0.5
			has_game_rule = {
				name = mongol_invasion
				value = random
			}
			NOT = { total_years_played = 100 }
		}
		modifier = {
			factor = 0.5
			has_game_rule = {
				name = mongol_invasion
				value = delayed_random
			}
			NOT = { total_years_played = 100 }
		}
		modifier = {
			factor = 0.5
			has_game_rule = {
				name = mongol_invasion
				value = random
			}
		}
		modifier = {
			factor = 0.6
			has_game_rule = {
				name = mongol_invasion
				value = delayed_random
			}
		}
	}
    immediate = {
        random_province = {
            limit = {
                OR = {
                    de_jure_liege_or_above = k_lakotah
                    de_jure_liege_or_above = k_big_sky
                    de_jure_liege_or_above = k_alberta
                    de_jure_liege_or_above = k_saskatchewan
                    de_jure_liege_or_above = k_manitoba
                }
            }
            save_global_event_target_as = joshua_origin
        }
        narrative_event = { id = gplains.2 }
    }
}

# Joshua Crossbearer appears
narrative_event = {
	id = gplains.2
	title = "EVTNAME_GPLAINS_2"
	desc = "EVTDESC_GPLAINS_2"
	major = yes
	hide_from = yes
	
	picture = GFX_evt_crossbearer
	border = "GFX_event_narrative_frame_war"
    
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = {
				NOT = { has_global_flag = joshua_arrival }
			}
			set_global_flag = joshua_arrival
			activate_title = { title = e_millenarian_host status = yes }
			
			if = {
				limit = {
					NOT = {
						e_millenarian_host = {
							holder_scope = { always = yes }
						}
					}
				}
				create_character = {
					name="Joshua"
					dynasty=2239451
                    give_nickname = nick_christus
					age = 29
					attributes = {
						martial=10
						diplomacy=8
						intrigue=8
						stewardship=9
					}
					health = 7
					fertility = 0.7
					religion="millenarian"
					culture=event_target:joshua_origin
					trait="brave"
					trait="ambitious"
					trait="diligent"
					trait="strategist"
					trait="cavalry_leader"
                    trait="conqueror"
					trait="exalted_warlord"
				}
				new_character = {
					set_character_flag = is_joshua
					save_global_event_target_as = invasion_leader
					if = {
						limit = {
							has_dlc = "Holy Fury"
							NOT = {
								any_owned_bloodline = {
									has_bloodline_flag = josh_cross_bloodline
								}
							}
						}
						create_bloodline = {
							type = genghis_khan
						}
					}
					e_millenarian_host = {
						set_title_landless = { title = THIS status = yes }
						set_title_flag = temporary_landless_title
						grant_title = PREV
					}
					set_defacto_liege = THIS
					if = {
						limit = {
							has_dlc = "Horse Lords"
						}
						set_government_type = nomadic_government
					}
					if = {
						limit = {
							NOT = { has_dlc = "Horse Lords" }
						}
						set_government_type = tribal_government
					}
					wealth = 12000
					prestige = 10000
					set_character_flag = horde_invader
					character_event = { id = gplains.3 } # Spawn generals
					character_event = { id = gplains.4 } # Spawn courtiers
					character_event = { id = gplains.5 } # Spawn troops
					character_event = { id = gplains.6 days = 1 } # Spawn population
				}
			}
			if = {
				limit = {
					has_dlc = "Horse Lords"
				}				
				event_target:joshua_origin = { # Invade province of origin
					kingdom = {
						e_millenarian_host = {
							holder_scope = {
								unsafe_war = {
									casus_belli = nomad_invasion
									target = PREVPREVPREV # The top liege of the owner of the province
									thirdparty_title = PREVPREV # The de jure kingdom
								}
							}
						}
					}
				}
			}
			if = {
				limit = {
					NOT = { has_dlc = "Horse Lords" }
				}				
				event_target:joshua_origin = {
					kingdom = {
						e_millenarian_host = {
							holder_scope = {
								unsafe_war = {
									casus_belli = tribal_invasion
									target = PREVPREVPREV # The top liege of the owner of the province
									thirdparty_title = PREVPREV # The de jure kingdom
								}
							}
						}
					}
				}
			}
		}
		any_playable_ruler = {
			limit = { ai = no }
			if = {
				limit = {
					OR = {
						has_power_omen_modifier_trigger = yes
						has_battle_omen_modifier_trigger = yes
					}
				}
				set_character_flag = negative_event_flag # For the scrying outcome calculation...
			}
		}
	}
	
	option = {
		name = "EVTOPTA_GPLAINS_2"
		trigger = {
			religion = millenarian
		}
	}
	option = {
		name = "EVTOPTB_GPLAINS_2"
		trigger = {
			capital_scope = {
				region = world_great_plainsUS
			}
			NOT = {
				religion = millenarian
			}
		}
	}
	option = {
		name = "EVTOPTC_GPLAINS_2"
		trigger = {
			NOT = {
				capital_scope = {
					region = world_great_plainsUS
				}
			}
			religion_group = christian
		}
	}
	option = {
		name = "EVTOPTD_GPLAINS_2"
		trigger = {
			NOR = {
				capital_scope = {
					region = world_great_plainsUS
				}
				religion_group = christian
				religion = millenarian
			}
		}
	}
	option = {
		name = "EVTOPTE_GPLAINS_2"
		trigger = {
			tooltip_info = yes
			trigger = {
				can_plays_as_invader_trigger = yes
			}
			play_as_invasion_leader_effect = yes
		}
	}
	
}

character_event = {
	id = gplains.7
	hide_window = yes
	
	is_triggered_only = yes
	
	immediate = {
		if = {
			limit = {
				NOT = { government = nomadic_government }
				NOT = { has_character_modifier = invading_population_halt }
			}
			remove_character_modifier = invading_population_halt
			break = yes
		}
		if = {
			limit = {
				NOT = { population_factor = 1 }
			}
			population = 80000
			if = {
				limit = {
					population = 350000
				}
				character_event = {
					id = gplains.8
					days = 365
				}
				break = yes
			}
		}
		repeat_event = { 
			id = gplains.7
			days = 365
		}
	}
}
