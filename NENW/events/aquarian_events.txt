character_event = { #destroys crystals on death, really because my way of making them ambiguous requires it, but insert handwave about crystals being personally attuned and draining over a lifetime
	id = aquarian.30121
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			has_artifact = aquarian_crystal
			has_artifact = aquarian_crystal_bad
		}
	}

	immediate = {
		any_artifact = {
			limit = { artifact_type = aquarian_crystal }
			destroy_artifact = yes
		}
		any_artifact = {
			limit = { artifact_type = aquarian_crystal_bad }
			destroy_artifact = yes
		}
	}
}
