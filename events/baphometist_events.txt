namespace = baph

# You summon your vassals to a Black Mass
character_event = {
	id = baph.600
	desc = EVTDESC_baph_600
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	hide_from = yes

	option = {
		name = EVTOPTA_TOG_600

		any_vassal = {
			limit = {
				block_general_event_trigger = no
				war = no
			}

			if = {
				limit = { religion = baphometist }

				character_event = {
					id = baph.601
					tooltip = EVTTOOLTIP_baph_601
				}

				character_event = {
					id = baph.605 # Blot begins
					days = 10
				}
			}
			else = {
				character_event = {
					id = baph.602
					tooltip = EVTTOOLTIP_baph_602
				}
			}
		}

		character_event = {
			id = baph.604 # Blot begins
			days = 10
		}
	}
}

# Non-Baphometists pay to skip Black Mass
character_event = {
	id = baph.602
	desc = EVTDESC_baph_602
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes
	hide_from = yes

	option = { # Pay
		name = EVTOPTA_TOG_602

		trigger = { wealth >= 5 }

		FROM = {
			show_scope_change = no

			character_event = {
				id = baph.603
				tooltip = EVTTOOLTIP_TOG_603
			}
		}

		wealth = -5
	}

	option = { # Don't pay
		name = EVTOPTB_TOG_602

		FROM = {
			show_scope_change = no

			character_event = {
				id = baph.636
				tooltip = EVTTOOLTIP_TOG_636
			}
		}

		FROM = {
			show_scope_change = no

			opinion = {
				name = opinion_refused_blot
				who = ROOT
			}

			any_vassal = {
				show_scope_change = no

				limit = { has_character_flag = attending_blot }

				opinion = {
					name = opinion_refused_blot
					who = ROOT
				}
			}
		}
	}
}

# Feast Begins (Vassal)
character_event = {
	id = baph.641
	desc = EVTDESC_baph_640
	picture = GFX_evt_viking_throneroom_oldgods
	border = GFX_event_normal_frame_religion

	is_triggered_only = yes

	ai = no

	option = {
		name = EVTOPTA_TOG_641

		custom_tooltip = { text = feast_begins }
	}
}

character_event = {
	id = baph.644

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		check_variable = {
			which = local_blot_feast_events
			value < 4
		}
	}

	immediate = {
		change_variable = {
			which = local_blot_feast_events
			value = 1
		}

		random_list = {
			10 = {
				trigger = {
					any_vassal = {
						trait = drunkard
						has_character_flag = blot_feast_vassal
					}

					NOT = { has_character_flag = feast_drunkard }
				}

				random_vassal = {
					limit = {
						trait = drunkard
						has_character_flag = blot_feast_vassal
					}

					character_event = { id = TOG.650 }
				}
			}

			10 = {
				trigger = {
					any_vassal = {
						trait = gluttonous
						has_character_flag = blot_feast_vassal
					}

					num_of_vassals >= 2
					NOT = { has_character_flag = feast_gluttonous }
				}

				random_vassal = {
					limit = {
						trait = gluttonous
						has_character_flag = blot_feast_vassal
					}

					character_event = { id = TOG.653 }
				}
			}

			10 = {
				trigger = {
					any_vassal = {
						trait = lunatic
						has_character_flag = blot_feast_vassal
					}

					NOT = { has_character_flag = feast_lunatic }
				}

				random_vassal = {
					limit = {
						trait = lunatic
						has_character_flag = blot_feast_vassal
					}

					character_event = { id = TOG.656 }
				}
			}

			10 = {
				trigger = {
					any_vassal = {
						trait = lustful
						is_female = no
						has_character_flag = blot_feast_vassal
					}

					NOT = { has_character_flag = feast_lustful }
				}

				random_vassal = {
					limit = {
						trait = lustful
						is_female = no
						has_character_flag = blot_feast_vassal
					}

					character_event = { id = TOG.658 }
				}
			}

			10 = {
				trigger = {
					any_vassal = {
						has_character_flag = blot_feast_vassal

						liege = {
							any_vassal = {
								has_character_flag = blot_feast_vassal

								opinion = {
									who = PREVPREV
									value >= 20
								}

								NOT = { character = PREVPREV }
							}
						}
					}

					NOT = { has_character_flag = feast_friendship }
				}

				random_vassal = {
					limit = {
						liege = {
							any_vassal = {
								opinion = {
									who = PREVPREV
									value >= 20
								}

								NOT = { character = PREVPREV }
							}
						}
					}

					character_event = { id = TOG.661 }
				}
			}

			10 = {
				trigger = {
					NOT = { has_character_flag = feast_homosexual }

					any_vassal = {
						trait = homosexual
						has_lover = no
						has_character_flag = blot_feast_vassal

						liege = {
							any_vassal = {
								trait = homosexual
								same_sex = PREVPREV
								has_lover = no
								has_character_flag = blot_feast_vassal

								opinion = {
									who = PREVPREV
									value >= 0
								}

								NOT = { character = PREVPREV }
							}
						}
					}
				}

				random_vassal = {
					limit = {
						trait = homosexual
						has_lover = no
						has_character_flag = blot_feast_vassal

						liege = {
							any_vassal = {
								trait = homosexual
								same_sex = PREVPREV
								has_lover = no
								has_character_flag = blot_feast_vassal

								opinion = {
									who = PREVPREV
									value >= 0
								}

								NOT = { character = PREVPREV }
							}
						}
					}

					character_event = { id = TOG.664 }
				}
			}
		}

		repeat_event = {
			id = baph.644
			days = 5
		}
	}

	fail_trigger_effect = {
		character_event = {
			id = baph.642 # Feast ends
			days = 7
		}
	}
}
