# Main corruption event upon an imperial title being inherited/usurped
# ROOT is the character, FROM is the title, FROMFROM is the old holder
character_event = {
	id = THEODEMO.1000 
	
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		AND = {
			FROMFROM = { always = yes } #If there is a FROMFROM
			FROM = {
				independent = yes
				temporary = no
				is_tribal = no
				is_nomadic = no
				is_primary_holder_title = yes
				government = theodemocratic
			}
			is_nomadic = no
			is_tribal = no
		}
	}
	
	immediate = {
		FROM = { save_event_target_as = government_balance_title }
		
		character_event = { id = THEODEMO.1001 days = 1 }
		
		# remove the corruption modifier from the former owner, if alive
		FROMFROM = {
			if = {
				limit = {
					is_alive = yes
				}
				remove_character_modifier = government_balance_0
				remove_character_modifier = government_balance_1
				remove_character_modifier = government_balance_2
				remove_character_modifier = government_balance_3
				remove_character_modifier = government_balance_4
				remove_character_modifier = government_balance_5
				remove_character_modifier = government_balance_6
				remove_character_modifier = government_balance_7
				remove_character_modifier = government_balance_8
				remove_character_modifier = government_balance_9
				remove_character_modifier = government_balance_10
			}
		}
	}
}
